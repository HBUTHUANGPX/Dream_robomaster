cmake_minimum_required(VERSION 3.22)
project(robomaster VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
include(cmake/NativeDependencies.cmake)
add_subdirectory(libs/sim)
foreach(module navigation duel cube)
  option(RM_BUILD_${module} "Build ${module}" ON)
  if(RM_BUILD_${module} AND EXISTS "${PROJECT_SOURCE_DIR}/modules/${module}/CMakeLists.txt")
    add_subdirectory(modules/${module})
  endif()
endforeach()
option(RM_BUILD_CAD "Build optional STEP converter (requires OpenCASCADE)" OFF)
if(RM_BUILD_CAD)
  add_subdirectory(tools/cad)
endif()
if(BUILD_TESTING)
  # Exercise process supervision without requiring the optional ROS1 stack.
  find_package(Threads REQUIRED)
  add_executable(fast_lio_supervisor_test integrations/fast_lio/replay.cpp)
  target_compile_definitions(fast_lio_supervisor_test PRIVATE RM_REPLAY_SUPERVISOR_SELF_TEST=1)
  target_link_libraries(fast_lio_supervisor_test PRIVATE Threads::Threads)
  add_test(NAME fast_lio_process_supervisor COMMAND fast_lio_supervisor_test)
  set_tests_properties(fast_lio_process_supervisor PROPERTIES TIMEOUT 15)
endif()
