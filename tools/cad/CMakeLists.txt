cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(robomaster_cad LANGUAGES CXX)
  include(CTest)
endif()
get_filename_component(RM_CAD_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(OCCT_ROOT "${RM_CAD_ROOT}/.deps/occt/usr" CACHE PATH "OCCT install prefix")
find_path(OCCT_INCLUDE_DIR Standard_Version.hxx HINTS "${OCCT_ROOT}/include/opencascade" PATH_SUFFIXES opencascade REQUIRED)
find_package(nlohmann_json 3.9 REQUIRED)
# Explicit tool dependencies avoid pulling visualization/OCAF into this offline converter.
set(OCCT_LIBRARIES "")
foreach(component TKSTEP TKSTEPAttr TKSTEP209 TKSTEPBase TKXSBase TKMesh TKShHealing TKTopAlgo TKBRep TKGeomAlgo TKGeomBase TKG3d TKG2d TKMath TKernel)
  find_library(OCCT_${component} NAMES ${component} HINTS "${OCCT_ROOT}/lib/${CMAKE_LIBRARY_ARCHITECTURE}" "${OCCT_ROOT}/lib" REQUIRED)
  list(APPEND OCCT_LIBRARIES "${OCCT_${component}}")
endforeach()
add_executable(rm_convert_cad convert_cad.cpp)
target_compile_features(rm_convert_cad PRIVATE cxx_std_20)
target_include_directories(rm_convert_cad SYSTEM PRIVATE "${OCCT_INCLUDE_DIR}")
target_link_libraries(rm_convert_cad PRIVATE ${OCCT_LIBRARIES} nlohmann_json::nlohmann_json)
target_compile_definitions(rm_convert_cad PRIVATE RM_CAD_SOURCE_ROOT="${RM_CAD_ROOT}")
get_filename_component(OCCT_LIBRARY_DIR "${OCCT_TKernel}" DIRECTORY)
set_target_properties(rm_convert_cad PROPERTIES BUILD_RPATH "${OCCT_LIBRARY_DIR}")
# Local extraction also contains transitive TBB/OCCT DSOs, with no system install.
if(UNIX AND NOT APPLE)
  target_link_options(rm_convert_cad PRIVATE "LINKER:--disable-new-dtags" "LINKER:-rpath-link,${OCCT_LIBRARY_DIR}")
endif()
if(BUILD_TESTING)
  add_test(NAME cad_coordinate_transforms COMMAND rm_convert_cad --self-test)
endif()
if(BUILD_TESTING AND EXISTS "${RM_CAD_ROOT}/assets/official/armor_am02.step")
  add_test(NAME cad_official_step_bounds COMMAND rm_convert_cad
    --root "${RM_CAD_ROOT}" --output "${RM_CAD_ROOT}/output/native-cad/ctest"
    --compare "${RM_CAD_ROOT}/assets/meshes/metadata.json")
  set_tests_properties(cad_official_step_bounds PROPERTIES TIMEOUT 180)
endif()
