cmake_minimum_required(VERSION 3.16)
project(fast_lio_replay LANGUAGES CXX)
# Noetic's generated message headers require the C++17 allocator interface.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(RM_REPLAY_BUILD_ROS "Build the ROS1 bag replay supervisor" ON)
include(CTest)
find_package(Threads REQUIRED)
if(RM_REPLAY_BUILD_ROS)
  find_package(catkin REQUIRED COMPONENTS roscpp rosbag sensor_msgs nav_msgs xmlrpcpp)
  add_executable(fast_lio_replay replay.cpp)
  target_include_directories(fast_lio_replay PRIVATE ${catkin_INCLUDE_DIRS})
  target_link_libraries(fast_lio_replay PRIVATE ${catkin_LIBRARIES} Threads::Threads)
  target_compile_options(fast_lio_replay PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS fast_lio_replay RUNTIME DESTINATION bin)
endif()
if(BUILD_TESTING)
  add_executable(fast_lio_supervisor_test replay.cpp)
  target_compile_definitions(fast_lio_supervisor_test PRIVATE RM_REPLAY_SUPERVISOR_SELF_TEST=1)
  target_compile_options(fast_lio_supervisor_test PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(fast_lio_supervisor_test PRIVATE Threads::Threads)
  add_test(NAME fast_lio_process_supervisor COMMAND fast_lio_supervisor_test)
  set_tests_properties(fast_lio_process_supervisor PROPERTIES TIMEOUT 15)
endif()
